# frozen_string_literal: true

require 'rails_helper'

RSpec.describe 'SafeMigrator reorder safety', type: :migration do
  # Regression test for: "Prevent reordered migration histories from changing the database"
  #
  # The safe migrator rejects a locally known migration hash when it appears after
  # a gap in the applied chain. This test directly proves that the ordering error
  # leaves both journal metadata and schema objects untouched — protecting against
  # accidental mutation before chain verification completes.

  let(:journal) { SafeMigrator::Journal.new }
  let(:schema_store) { SafeMigrator::SchemaStore.new }
  let(:verifier) { SafeMigrator::ChainVerifier.new(journal: journal) }
  let(:runner) { SafeMigrator::Runner.new(journal: journal, schema_store: schema_store, verifier: verifier) }

  # Simulated migration hashes representing a history
  let(:hash_a) { 'aaaa1111' }
  let(:hash_b) { 'bbbb2222' }
  let(:hash_c) { 'cccc3333' }
  let(:hash_d) { 'dddd4444' }
  let(:hash_e) { 'eeee5555' }

  # Simulated migration class
  let(:migration_class) do
    Class.new do
      def self.up(connection)
        connection.execute("CREATE TABLE IF NOT EXISTS reorder_test_table (id INTEGER PRIMARY KEY)")
      end

      def self.down(connection)
        connection.execute("DROP TABLE IF EXISTS reorder_test_table")
      end
    end
  end

  before do
    # Clean slate for each test
    journal.clear!
    schema_store.clear!
  end

  describe 'when migration history is reordered with a gap' do
    # Scenario: Applied chain is [A, B, C], but local history claims [A, C, B, D]
    # This creates a gap: after A, the next applied is C (skipping B in local view),
    # then B appears after C — a reorder that should be rejected.

    before do
      # Simulate already-applied chain: A -> B -> C
      journal.record!(hash: hash_a, version: '001', applied_at: 1.day.ago)
      journal.record!(hash: hash_b, version: '002', applied_at: 2.days.ago)
      journal.record!(hash: hash_c, version: '003', applied_at: 3.days.ago)
    end

    it 'rejects the reordered local history without mutating journal metadata' do
      # Local history claims order: A -> C -> B -> D (reordered: B and C swapped)
      local_history = [
        { hash: hash_a, version: '001' },
        { hash: hash_c, version: '003' }, # C appears before B — gap in applied chain
        { hash: hash_b, version: '002' }, # B appears after C — reorder detected
        { hash: hash_d, version: '004' }
      ]

      # Capture journal state before attempt
      journal_before = journal.all_entries.map(&:to_h)
      journal_count_before = journal.count

      # Attempt to run — should raise due to reorder/gap detection
      expect {
        runner.run(local_history: local_history, migration_class: migration_class)
      }.to raise_error(SafeMigrator::ChainOrderError, /gap|reorder|order/i)

      # CRITICAL ASSERTION: Journal metadata must remain untouched
      journal_after = journal.all_entries.map(&:to_h)
      expect(journal_after).to eq(journal_before),
        'Journal metadata was mutated despite chain order error'
      expect(journal.count).to eq(journal_count_before),
        'Journal entry count changed despite chain order error'

      # Verify no new entries were added
      expect(journal.find_by_hash(hash_d)).to be_nil,
        'Unapplied migration was recorded in journal despite rejection'
    end

    it 'rejects the reordered local history without creating schema objects' do
      local_history = [
        { hash: hash_a, version: '001' },
        { hash: hash_c, version: '003' },
        { hash: hash_b, version: '002' },
        { hash: hash_d, version: '004' }
      ]

      # Capture schema state before attempt
      tables_before = schema_store.list_tables

      # Attempt to run — should raise
      expect {
        runner.run(local_history: local_history, migration_class: migration_class)
      }.to raise_error(SafeMigrator::ChainOrderError)

      # CRITICAL ASSERTION: Schema objects must remain untouched
      tables_after = schema_store.list_tables
      expect(tables_after).to eq(tables_before),
        'Schema objects were created despite chain order error'

      # Verify the migration's side effect did not occur
      expect(schema_store.table_exists?('reorder_test_table')).to be false,
        'Migration DDL executed despite chain order rejection'
    end

    it 'does not partially apply any migration before detecting the reorder' do
      local_history = [
        { hash: hash_a, version: '001' },
        { hash: hash_c, version: '003' },
        { hash: hash_b, version: '002' },
        { hash: hash_d, version: '004' }
      ]

      # Attempt to run
      expect {
        runner.run(local_history: local_history, migration_class: migration_class)
      }.to raise_error(SafeMigrator::ChainOrderError)

      # Verify chain verification happens BEFORE any mutation
      # The runner must verify the entire chain before applying anything
      expect(journal.find_by_hash(hash_d)).to be_nil
      expect(schema_store.table_exists?('reorder_test_table')).to be false
    end
  end

  describe 'when a known hash appears after a gap in the applied chain' do
    # Scenario: Applied chain is [A, B], but local history claims [A, C, B]
    # C is unknown locally but appears between A and B — creating a gap
    # Then B (a known hash) appears after the gap — should be rejected

    before do
      journal.record!(hash: hash_a, version: '001', applied_at: 1.day.ago)
      journal.record!(hash: hash_b, version: '002', applied_at: 2.days.ago)
    end

    it 'rejects when a known hash appears after an unknown gap' do
      # Local history: A -> C (unknown) -> B (known, but after gap)
      local_history = [
        { hash: hash_a, version: '001' },
        { hash: hash_c, version: '003' }, # Unknown hash creates gap
        { hash: hash_b, version: '002' }  # Known hash after gap — rejected
      ]

      journal_before = journal.all_entries.map(&:to_h)
      tables_before = schema_store.list_tables

      expect {
        runner.run(local_history: local_history, migration_class: migration_class)
      }.to raise_error(SafeMigrator::ChainOrderError)

      # Prove no mutation occurred
      expect(journal.all_entries.map(&:to_h)).to eq(journal_before)
      expect(schema_store.list_tables).to eq(tables_before)
    end
  end

  describe 'chain verification timing' do
    it 'verifies the entire chain before applying any migration' do
      # Apply initial state
      journal.record!(hash: hash_a, version: '001', applied_at: 1.day.ago)

      # Local history with reorder at the end
      local_history = [
        { hash: hash_a, version: '001' },
        { hash: hash_b, version: '002' },
        { hash: hash_d, version: '004' }, # Skip C
        { hash: hash_c, version: '003' }  # C appears after D — reorder
      ]

      # Track whether migration.up was called
      up_called = false
      tracking_migration = Class.new do
        define_singleton_method(:up) do |connection|
          up_called = true
          connection.execute("CREATE TABLE IF NOT EXISTS should_not_exist (id INTEGER)")
        end
      end

      expect {
        runner.run(local_history: local_history, migration_class: tracking_migration)
      }.to raise_error(SafeMigrator::ChainOrderError)

      # Prove migration.up was never called — verification happens first
      expect(up_called).to be false,
        'Migration up() was called before chain verification completed'
    end
  end

  describe 'journal immutability under error conditions' do
    it 'preserves journal entries even when multiple errors occur' do
      journal.record!(hash: hash_a, version: '001', applied_at: 1.day.ago)
      journal.record!(hash: hash_b, version: '002', applied_at: 2.days.ago)

      local_history = [
        { hash: hash_a, version: '001' },
        { hash: hash_c, version: '003' }, # Gap
        { hash: hash_b, version: '002' }  # Reorder
      ]

      # Attempt multiple times to ensure no cumulative mutation
      3.times do
        expect {
          runner.run(local_history: local_history, migration_class: migration_class)
        }.to raise_error(SafeMigrator::ChainOrderError)
      end

      # Journal must still have exactly the original 2 entries
      expect(journal.count).to eq(2)
      expect(journal.find_by_hash(hash_a)).to be_present
      expect(journal.find_by_hash(hash_b)).to be_present
      expect(journal.find_by_hash(hash_c)).to be_nil
    end
  end
end

# frozen_string_literal: true

require 'rails_helper'

RSpec.describe 'SafeMigrator reorder safety', type: :migration do
  # Regression test for: "Prevent reordered migration histories from changing the database"
  #
  # The safe migrator rejects a locally known migration hash when it appears after
  # a gap in the applied chain. This test directly proves that the ordering error
  # leaves both journal metadata and schema objects untouched — protecting against
  # accidental mutation before chain verification completes.

  let(:journal) { SafeMigrator::Journal.new }
  let(:schema_store) { SafeMigrator::SchemaStore.new }
  let(:verifier) { SafeMigrator::ChainVerifier.new(journal: journal) }
  let(:runner) { SafeMigrator::Runner.new(journal: journal, schema_store: schema_store, verifier: verifier) }

  # Simulated migration hashes representing a history
  let(:hash_a) { 'aaaa1111' }
  let(:hash_b) { 'bbbb2222' }
  let(:hash_c) { 'cccc3333' }
  let(:hash_d) { 'dddd4444' }

  # Simulated migration class that would create a table if executed
  let(:side_effect_migration) do
    Class.new do
      def self.up(connection)
        connection.execute("CREATE TABLE IF NOT EXISTS reorder_test_table (id INTEGER PRIMARY KEY)")
      end

      def self.down(connection)
        connection.execute("DROP TABLE IF EXISTS reorder_test_table")
      end
    end
  end

  before do
    journal.clear!
    schema_store.clear!
  end

  describe 'when a locally known hash appears after a gap in the applied chain' do
    # Scenario: Applied chain is [A, B, C], but local history claims [A, C, B, D]
    # This creates a gap: after A, the next applied is C (skipping B in local view),
    # then B (a known hash) appears after C — a reorder that must be rejected.

    before do
      # Simulate already-applied chain: A -> B -> C
      journal.record!(hash: hash_a, version: '001', applied_at: 1.day.ago)
      journal.record!(hash: hash_b, version: '002', applied_at: 2.days.ago)
      journal.record!(hash: hash_c, version: '003', applied_at: 3.days.ago)
    end

    it 'rejects the reorder without mutating journal metadata' do
      # Local history claims order: A -> C -> B -> D (B and C swapped — reorder)
      local_history = [
        { hash: hash_a, version: '001' },
        { hash: hash_c, version: '003' }, # C appears before B in local view
        { hash: hash_b, version: '002' }, # B appears after C — reorder detected
        { hash: hash_d, version: '004' }
      ]

      # Capture journal state before attempt
      journal_snapshot_before = journal.all_entries.map(&:to_h)
      journal_count_before = journal.count

      # Attempt to run — must raise due to reorder/gap detection
      expect {
        runner.run(local_history: local_history, migration_class: side_effect_migration)
      }.to raise_error(SafeMigrator::ChainOrderError)

      # CRITICAL: Journal metadata must remain untouched
      journal_snapshot_after = journal.all_entries.map(&:to_h)
      expect(journal_snapshot_after).to eq(journal_snapshot_before),
        'Journal metadata was mutated despite chain order error'
      expect(journal.count).to eq(journal_count_before),
        'Journal entry count changed despite chain order error'

      # Verify no new entries were added for the unapplied migration
      expect(journal.find_by_hash(hash_d)).to be_nil,
        'Unapplied migration was recorded in journal despite rejection'
    end

    it 'rejects the reorder without creating schema objects' do
      local_history = [
        { hash: hash_a, version: '001' },
        { hash: hash_c, version: '003' },
        { hash: hash_b, version: '002' },
        { hash: hash_d, version: '004' }
      ]

      # Capture schema state before attempt
      tables_before = schema_store.list_tables

      expect {
        runner.run(local_history: local_history, migration_class: side_effect_migration)
      }.to raise_error(SafeMigrator::ChainOrderError)

      # CRITICAL: Schema objects must remain untouched
      tables_after = schema_store.list_tables
      expect(tables_after).to eq(tables_before),
        'Schema objects were created despite chain order error'

      # Verify the migration DDL did not execute
      expect(schema_store.table_exists?('reorder_test_table')).to be false,
        'Migration DDL executed despite chain order rejection'
    end

    it 'verifies the entire chain before applying any migration' do
      local_history = [
        { hash: hash_a, version: '001' },
        { hash: hash_c, version: '003' },
        { hash: hash_b, version: '002' },
        { hash: hash_d, version: '004' }
      ]

      # Track whether migration.up was called
      up_called = false
      tracking_migration = Class.new do
        define_singleton_method(:up) do |_connection|
          up_called = true
        end
      end

      expect {
        runner.run(local_history: local_history, migration_class: tracking_migration)
      }.to raise_error(SafeMigrator::ChainOrderError)

      # Prove migration.up was never called — verification happens before any mutation
      expect(up_called).to be false,
        'Migration up() was called before chain verification completed'
    end
  end

  describe 'when a known hash appears after an unknown gap' do
    # Scenario: Applied chain is [A, B], but local history claims [A, C, B]
    # C is unknown in the journal but appears between A and B — creating a gap.
    # Then B (a known hash) appears after the gap — must be rejected.

    before do
      journal.record!(hash: hash_a, version: '001', applied_at: 1.day.ago)
      journal.record!(hash: hash_b, version: '002', applied_at: 2.days.ago)
    end

    it 'rejects without mutating journal or schema' do
      local_history = [
        { hash: hash_a, version: '001' },
        { hash: hash_c, version: '003' }, # Unknown hash creates gap
        { hash: hash_b, version: '002' }  # Known hash after gap — rejected
      ]

      journal_before = journal.all_entries.map(&:to_h)
      tables_before = schema_store.list_tables

      expect {
        runner.run(local_history: local_history, migration_class: side_effect_migration)
      }.to raise_error(SafeMigrator::ChainOrderError)

      expect(journal.all_entries.map(&:to_h)).to eq(journal_before),
        'Journal was mutated when known hash appeared after gap'
      expect(schema_store.list_tables).to eq(tables_before),
        'Schema was mutated when known hash appeared after gap'
    end
  end

  describe 'journal immutability under repeated error conditions' do
    it 'preserves journal entries across multiple rejection attempts' do
      journal.record!(hash: hash_a, version: '001', applied_at: 1.day.ago)
      journal.record!(hash: hash_b, version: '002', applied_at: 2.days.ago)

      local_history = [
        { hash: hash_a, version: '001' },
        { hash: hash_c, version: '003' }, # Gap
        { hash: hash_b, version: '002' }  # Reorder
      ]

      # Attempt multiple times to ensure no cumulative mutation
      3.times do
        expect {
          runner.run(local_history: local_history, migration_class: side_effect_migration)
        }.to raise_error(SafeMigrator::ChainOrderError)
      end

      # Journal must still have exactly the original 2 entries
      expect(journal.count).to eq(2)
      expect(journal.find_by_hash(hash_a)).to be_present
      expect(journal.find_by_hash(hash_b)).to be_present
      expect(journal.find_by_hash(hash_c)).to be_nil
    end
  end

  describe 'schema immutability when reorder is detected mid-chain' do
    it 'does not partially apply earlier migrations before detecting reorder' do
      # Start with only A applied
      journal.record!(hash: hash_a, version: '001', applied_at: 1.day.ago)

      # Local history: A -> B -> D -> C (reorder at the end)
      local_history = [
        { hash: hash_a, version: '001' },
        { hash: hash_b, version: '002' },
        { hash: hash_d, version: '004' }, # Skip C — gap
        { hash: hash_c, version: '003' }  # C appears after D — reorder
      ]

      tables_before = schema_store.list_tables

      expect {
        runner.run(local_history: local_history, migration_class: side_effect_migration)
      }.to raise_error(SafeMigrator::ChainOrderError)

      # No schema changes should have occurred — not even for B
      expect(schema_store.list_tables).to eq(tables_before),
        'Schema was partially mutated before reorder detection'
      expect(schema_store.table_exists?('reorder_test_table')).to be false
    end
  end
end
