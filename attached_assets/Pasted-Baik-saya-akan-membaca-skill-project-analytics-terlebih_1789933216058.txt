# frozen_string_literal: true

module Import
  class FunnelEvent < ApplicationRecord
    self.table_name = 'import_funnel_events'

    belongs_to :import
    belongs_to :user, optional: true

    # Event types representing the import funnel stages
    enum :event_type, {
      started: 0,
      validated: 1,
      validation_failed: 2,
      previewed: 3,
      committed: 4,
      commit_failed: 5,
      cancelled: 6
    }

    # Bounded failure categories for commit failures
    # These distinguish infrastructure/authorization errors from user abandonment
    enum :failure_category, {
      unknown: 0,
      infrastructure_error: 1,
      authorization_error: 2,
      validation_error: 3,
      timeout_error: 4,
      conflict_error: 5
    }, _prefix: :failure

    validates :event_type, presence: true
    validates :failure_category, presence: true, if: -> { commit_failed? }

    scope :commits_failed, -> { where(event_type: :commit_failed) }
    scope :with_failure_category, ->(category) { commits_failed.where(failure_category: category) }

    # Helper method to record commit failures with appropriate category
    def self.record_commit_failure(import:, user: nil, error: nil, metadata: {})
      failure_cat = categorize_error(error)
      
      create!(
        import: import,
        user: user,
        event_type: :commit_failed,
        failure_category: failure_cat,
        metadata: metadata.merge(
          error_class: error&.class&.name,
          error_message: error&.message&.truncate(500),
          failure_category: failure_cat
        )
      )
    end

    # Categorize errors into bounded failure categories
    def self.categorize_error(error)
      return :unknown if error.nil?

      case error
      when ActiveRecord::RecordInvalid, ActiveRecord::RecordNotSaved
        :validation_error
      when ActiveRecord::StaleObjectError, ActiveRecord::RecordNotUnique
        :conflict_error
      when Timeout::Error, Net::ReadTimeout, Net::OpenTimeout
        :timeout_error
      when Pundit::NotAuthorizedError, CanCan::AccessDenied, StandardError
        # Check if it's an authorization error by class name or message
        if authorization_error?(error)
          :authorization_error
        elsif infrastructure_error?(error)
          :infrastructure_error
        else
          :unknown
        end
      else
        :unknown
      end
    end

    def self.authorization_error?(error)
      error.class.name.include?('Authorization') ||
        error.class.name.include?('NotAuthorized') ||
        error.class.name.include?('AccessDenied') ||
        error.class.name.include?('Forbidden') ||
        error.class.name.include?('Unauthorized') ||
        error.message.to_s.match?(/not authorized|access denied|forbidden|unauthorized/i)
    end

    def self.infrastructure_error?(error)
      error.class.name.include?('Connection') ||
        error.class.name.include?('Timeout') ||
        error.class.name.include?('Socket') ||
        error.class.name.include?('Redis') ||
        error.class.name.include?('Database') ||
        error.message.to_s.match?(/connection|timeout|socket|redis|database/i)
    end
  end
end

module Import
  class CommitService
    attr_reader :import, :user, :result

    def initialize(import:, user:)
      @import = import
      @user = user
      @result = nil
    end

    def call
      return failure(:already_committed) if import.committed?
      return failure(:not_validated) unless import.validated?

      ActiveRecord::Base.transaction do
        import.update!(status: :committing, committed_at: Time.current, committed_by: user.id)
        
        process_import_data
        
        import.update!(status: :committed)
        track_commit_success
        
        @result = success(import)
      rescue StandardError => e
        handle_commit_error(e)
        raise ActiveRecord::Rollback
      end

      @result
    rescue StandardError => e
      @result ||= failure(:commit_error, error: e)
    end

    private

    def process_import_data
      # Process the actual import logic here
      # This is a placeholder for the actual implementation
      import.process_data!
    end

    def track_commit_success
      Import::FunnelEvent.create!(
        import: import,
        user: user,
        event_type: :committed,
        metadata: {
          committed_at: import.committed_at,
          records_count: import.records_count
        }
      )
    end

    def handle_commit_error(error)
      import.update!(status: :commit_failed, error_message: error.message)
      
      Import::FunnelEvent.record_commit_failure(
        import: import,
        user: user,
        error: error,
        metadata: {
          attempted_at: Time.current,
          records_count: import.records_count
        }
      )
      
      Rails.logger.error("[ImportCommitService] Commit failed for import #{import.id}: #{error.class} - #{error.message}")
    end

    def success(import)
      ServiceResult.new(success: true, value: import)
    end

    def failure(reason, error: nil)
      ServiceResult.new(success: false, error: reason, value: error)
    end
  end
end

# frozen_string_literal: true

module Import
  class AnalyticsService
    def initialize(import: nil, user: nil, date_range: nil)
      @import = import
      @user = user
      @date_range = date_range || (30.days.ago..Time.current)
    end

    def funnel_summary
      events = base_events
      
      {
        started: events.where(event_type: :started).count,
        validated: events.where(event_type: :validated).count,
        validation_failed: events.where(event_type: :validation_failed).count,
        previewed: events.where(event_type: :previewed).count,
        committed: events.where(event_type: :committed).count,
        commit_failed: events.where(event_type: :commit_failed).count,
        cancelled: events.where(event_type: :cancelled).count
      }
    end

    def commit_failure_breakdown
      commit_failures = base_events.commits_failed
      
      {
        total: commit_failures.count,
        by_category: commit_failures.group(:failure_category).count.transform_keys do |key|
          Import::FunnelEvent.failure_categories.key(key)
        end,
        infrastructure_errors: commit_failures.with_failure_category(:infrastructure_error).count,
        authorization_errors: commit_failures.with_failure_category(:authorization_error).count,
        validation_errors: commit_failures.with_failure_category(:validation_error).count,
        timeout_errors: commit_failures.with_failure_category(:timeout_error).count,
        conflict_errors: commit_failures.with_failure_category(:conflict_error).count,
        unknown_errors: commit_failures.with_failure_category(:unknown).count
      }
    end

    def recovery_analysis
      # Analyze whether administrators return after validation failures
      validation_failures = base_events.where(event_type: :validation_failed)
      
      recovery_data = validation_failures.map do |failure|
        subsequent_validations = base_events.where(
          import_id: failure.import_id,
          event_type: :validated
        ).where('occurred_at > ?', failure.occurred_at)
        
        {
          import_id: failure.import_id,
          failed_at: failure.occurred_at,
          recovered: subsequent_validations.exists?,
          recovery_time: subsequent_validations.first&.occurred_at
        }
      end

      {
        total_failures: validation_failures.count,
        recovered: recovery_data.count { |r| r[:recovered] },
        not_recovered: recovery_data.count { |r| !r[:recovered] },
        recovery_rate: validation_failures.count.positive? ? 
          (recovery_data.count { |r| r[:recovered] }.to_f / validation_failures.count * 100).round(2) : 0
      }
    end

    def abandonment_vs_error_analysis
      # Distinguish between user abandonment and system errors
      cancellations = base_events.where(event_type: :cancelled).count
      commit_failures = base_events.where(event_type: :commit_failed).count
      
      {
        cancellations: cancellations,
        commit_failures: commit_failures,
        commit_failure_breakdown: commit_failure_breakdown,
        abandonment_rate: calculate_abandonment_rate,
        error_rate: calculate_error_rate
      }
    end

    private

    def base_events
      scope = Import::FunnelEvent.where(occurred_at: @date_range)
      scope = scope.where(import: @import) if @import
      scope = scope.where(user: @user) if @user
      scope
    end

    def calculate_abandonment_rate
      total_started = base_events.where(event_type: :started).count
      return 0 if total_started.zero?

      cancellations = base_events.where(event_type: :cancelled).count
      (cancellations.to_f / total_started * 100).round(2)
    end

    def calculate_error_rate
      total_started = base_events.where(event_type: :started).count
      return 0 if total_started.zero?

      commit_failures = base_events.where(event_type: :commit_failed).count
      (commit_failures.to_f / total_started * 100).round(2)
    end
  end
end
